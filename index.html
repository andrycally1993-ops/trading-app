async function fetchRealBalance() { const apiKey = "f2f94d98-56b4-4c5a-b363-980a24b16839"; const secretKey = "F4BC6222EC9594F4DB889A1B5532"; const passphrase = "King_9396"; try { const timestamp = new Date().toISOString(); const path = '/api/v5/account/balance'; const method = 'GET'; const message = timestamp + method + path; // Firma HMAC-SHA256 calcolata nativamente nel browser const enc = new TextEncoder(); const key = await crypto.subtle.importKey( "raw", enc.encode(secretKey), { name: "HMAC", hash: "SHA-256" }, false, ["sign"] ); const signatureBuffer = await crypto.subtle.sign("HMAC", key, enc.encode(message)); const base64Signature = btoa(String.fromCharCode(...new Uint8Array(signatureBuffer))); // Chiamata diretta a OKX bypassando il cloud di Render const response = await fetch("https://www.okx.com" + path, { method: "GET", headers: { "OK-ACCESS-KEY": apiKey, "OK-ACCESS-SIGN": base64Signature, "OK-ACCESS-TIMESTAMP": timestamp, "OK-ACCESS-PASSPHRASE": passphrase, "Content-Type": "application/json" } }); const data = await response.json(); if (data.code === '0' && data.data && data.data.length > 0) { const balanceInfo = data.data[0]; let usdcBalance = "0.00"; for (let detail of balanceInfo.details || []) { if (detail.ccy === 'USDC') { usdcBalance = detail.availEq || detail.cashBal || "0.00"; break; } } // Aggiorna la grafica della web app con i tuoi veri fondi document.querySelector('.balance-display').innerText = `€${parseFloat(usdcBalance).toLocaleString('it-IT', {minimumFractionDigits: 2})}`; console.log("Saldo reale USDC caricato con successo dal browser:", usdcBalance); } } catch (err) { console.error("Errore nella lettura diretta del saldo:", err); } } // Avvia la sincronizzazione automatica ogni 10 secondi setInterval(fetchRealBalance, 10000); fetchRealBalance();
